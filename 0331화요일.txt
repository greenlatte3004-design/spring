@mcp.tool()
async def fetch_notion_content() -> str:
    """노션 페이지의 텍스트 내용을 읽어옵니다."""
    url = f"https://api.notion.com/v1/blocks/{NOTION_PAGE_ID}/children"
    headers = {
        "Authorization": f"Bearer {NOTION_TOKEN}",  #OAuth2방식 & Bearer방식 
        "Notion-Version": "2026-03-11",
    }
    response = requests.get(url, headers=headers)
    data = response.json()
    lines = [
        b["paragraph"]["rich_text"][0]["plain_text"]
        for b in data.get("results", [])
        if b["type"] == "paragraph" and b["paragraph"]["rich_text"]
    ]
    return "\n".join(lines) if lines else "내용 없음"

@mcp.tool()
async def add_image_to_notion(image_url: str) -> str:
    """
    노션 페이지에 외부 이미지(URL)를 추가합니다.
    image_url: 추가할 이미지의 공개 HTTP URL
    """
    url = f"https://api.notion.com/v1/blocks/{NOTION_PAGE_ID}/children"
    headers = {
        "Authorization": f"Bearer {NOTION_TOKEN}",
        "Notion-Version": "2026-03-11",
        "Content-Type": "application/json",
    }
    payload = {
        "children": [
            {
                "object": "block",
                "type": "image",
                "image": {"type": "external", "external": {"url": image_url}},
            }
        ]
    }
    response = requests.patch(url, headers=headers, json=payload)
    if response.status_code == 200:
        return f"이미지 삽입 성공: {image_url}"
    return f"이미지 삽입 실패: {response.text}"
